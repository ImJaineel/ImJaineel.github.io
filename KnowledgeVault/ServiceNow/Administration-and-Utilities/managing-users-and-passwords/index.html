Managing Users & Passwords in ServiceNow

This guide covers programmatic ways to manage users, update passwords, and handle active sessions across ServiceNow instances using both Background Scripts (GlideRecord) and NodeJS (OOTB REST APIs).

1. Creating a User with Elevated Privileges (Background Script)​

Sometimes you need to programmatically create a user and assign them specific roles (like admin and security_admin) along with custom user preferences.

APIs Used:

  • GlideRecord('sys_user') - For creating the user record.
  • GlideSecurityManager.get().enableElevatedRole() - Required to assign security_admin.
  • sys_user_has_role and sys_user_preference tables for relationships.
javascript
// Create the user record
var grSysUser = new GlideRecord("sys_user");
grSysUser.initialize();
grSysUser.setNewGuidValue('jaineelpetiwale'); // Set custom sys_id (optional, usually auto-generated)
grSysUser.setDisplayValue('user_name', 'jaineel.petiwale'); 
grSysUser.setDisplayValue('first_name', 'Jaineel'); 
grSysUser.setDisplayValue('last_name', 'Petiwale');
grSysUser.setDisplayValue('email', 'Jaineel2001@gmail.com'); 
grSysUser.setDisplayValue('title', 'ServiceNow Developer'); 
grSysUser.setDisplayValue('department', 'IT'); 
grSysUser.setValue('preferred_language','en'); 
grSysUser.setDisplayValue('gender','Male'); 
grSysUser.setDisplayValue('identity_type', 'human'); 
grSysUser.setDisplayValue('manager','admin'); 
grSysUser.setValue('active', true); 
grSysUser.setDisplayValue('user_password', 'MySecretPassword123'); // Set the password
grSysUser.setValue('password_needs_reset', false); 

// System fields (sys_created_on/by) can be overridden if needed
grSysUser.setValue('sys_created_on', '2001-03-21 05:30:00'); 
grSysUser.setValue('sys_created_by', 'jaineel.petiwale'); 
grSysUser.autoSysFields(); // Ensure sys fields are handled
var userSysId = grSysUser.insert();

// Set the password using GlideSystem, assign roles, and set preferences
if (userSysId) {
    // Assign admin and security_admin roles
    GlideSecurityManager.get().enableElevatedRole('security_admin');
    addRole(userSysId, "admin");
    addRole(userSysId, "security_admin");
    GlideSecurityManager.get().disableElevatedRole('security_admin');
    gs.info("Roles 'admin' and 'security_admin' assigned to the user.");

    // Set "Always Show Top Navigation" preference for UI16/Polaris
    setUserPreference(userSysId, "glide.ui.polaris.ui16_tabs_inside_polaris", true);
    gs.info("User preference 'Always Show Top Navigation' set to true.");
} else {
    gs.info("User creation failed.");
}

// Function to assign a role to a user
function addRole(userSysId, roleName) {
    var grSysUserHasRole = new GlideRecord("sys_user_has_role");
    grSysUserHasRole.initialize();
    grSysUserHasRole.setValue('user', userSysId);
    grSysUserHasRole.role.setDisplayValue(roleName);
    grSysUserHasRole.insert();
}

// Function to set user preference
function setUserPreference(userSysId, name, value) {
    var grSysUserPreference = new GlideRecord("sys_user_preference");
    grSysUserPreference.initialize();
    grSysUserPreference.setValue('user', userSysId);
    grSysUserPreference.setDisplayValue('name', name);
    grSysUserPreference.setValue('value', value);
    grSysUserPreference.insert();
}

2. Quickly Updating a Local User Password (Background Script)​

If you just need to reset a password for a specific user using their sys_id, you can simply update the user_password field via setDisplayValue.

APIs Used:

  • GlideRecord .get() and .update()
javascript
var grSysUser = new GlideRecord('sys_user');
if (grSysUser.get('PUT_USER_SYS_ID_HERE')) {
    grSysUser.user_password.setDisplayValue('new_password_here');
    grSysUser.update();
}

3. Bulk Updating User Passwords across Instances (NodeJS)​

Using the NodeJS script below, you can generate a strong randomized password and update a non-admin user's password across multiple ServiceNow instances using the Table API.

APIs Used:

  • Table API (/api/now/table/sys_user/{sys_id}?sysparm_input_display_value=true)
  • Native NodeJS https and crypto modules
javascript
import https from 'https';
import { randomBytes } from 'crypto';
import { URL } from 'url';

// List of ServiceNow instances (e.g., 'dev12345')
const instances = ['dev225514'];

// Service account credentials
const username = 'your_username';
const password = 'your_password';
const userSysId = 'PUT_USER_SYS_ID_HERE';

// Generate a secure 20-character alphanumeric password
function generatePassword(length = 20) {
  const charset = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
  const bytes = randomBytes(length);
  return Array.from(bytes).map(b => charset[b % charset.length]).join('');
}

// Make an HTTPS PATCH request to update the password via Table API
function updatePassword(instanceUrl, userSysId, newPassword, username, password) {
  return new Promise((resolve, reject) => {
    // Note: sysparm_input_display_value=true is crucial when setting passwords directly
    const apiUrl = new URL(`/api/now/table/sys_user/${userSysId}?sysparm_input_display_value=true`, `https://${instanceUrl}.service-now.com`);
    const body = JSON.stringify({ user_password: newPassword });
    const auth = Buffer.from(`${username}:${password}`).toString('base64');

    const options = {
      method: 'PATCH',
      headers: {
        'Authorization': `Basic ${auth}`,
        'Content-Type': 'application/json',
        'Content-Length': Buffer.byteLength(body)
      }
    };

    const req = https.request(apiUrl, options, res => {
      let data = '';
      res.on('data', chunk => data += chunk);
      res.on('end', () => {
        if (res.statusCode >= 200 && res.statusCode < 300) {
          resolve(`Success: ${instanceUrl}`);
        } else {
          reject(`Failed on ${instanceUrl}: ${res.statusCode} - ${data}`);
        }
      });
    });

    req.on('error', reject);
    req.write(body);
    req.end();
  });
}

(async () => {
  const newPassword = generatePassword();
  console.log(`Generated Password: ${newPassword}`);

  for (const instance of instances) {
    try {
      const result = await updatePassword(instance, userSysId, newPassword, username, password);
      console.log(result);
    } catch (err) {
      console.error(err);
    }
  }
})();

4. Forcing User Logout (Background Script)​

If a user is compromised or needs their session invalidated immediately across all nodes, use the GlideSessions API.

APIs Used:

  • GlideSessions.lockOutSessionsInAllNodes(username)
javascript
// Forcefully invalidates all active sessions for the specified user
GlideSessions.lockOutSessionsInAllNodes("<username>");
