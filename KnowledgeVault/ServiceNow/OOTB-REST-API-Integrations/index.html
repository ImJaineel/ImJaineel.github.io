OOTB REST API Integrations (NodeJS)

ServiceNow provides powerful Out-Of-The-Box (OOTB) REST APIs that allow you to interact with the platform externally. This guide demonstrates how to use NodeJS to perform bulk updates and search codebase assets externally.

1. Bulk Updating Records (Table API)​

While you can run background scripts natively, sometimes you need to manipulate data from an external CI/CD pipeline or integration server. The script below uses the standard Table API to fetch and update records in bulk.

APIs Used:

  • GET /api/now/table/{table_name} - Fetches a list of sys_ids matching an encoded query.
  • PATCH /api/now/table/{table_name}/{sys_id} - Updates individual records based on the fetched IDs.
javascript
import fetch from 'node-fetch'; // Requires node-fetch if Node < 18

const INSTANCE = 'your_instance';
const USERNAME = 'admin_user';
const PASSWORD = 'admin_password';

const TABLE = 'incident';
const ENCODED_QUERY = 'active=true^state=1'; // Example: Active New Incidents
const LIMIT = 100;

const authHeader = 'Basic ' + Buffer.from(`${USERNAME}:${PASSWORD}`).toString('base64');

// 1. Fetch sys_ids
async function fetchSysIds() {
  const url = `https://${INSTANCE}.service-now.com/api/now/table/${TABLE}?sysparm_query=${ENCODED_QUERY}&sysparm_fields=sys_id&sysparm_limit=${LIMIT}`;
  const response = await fetch(url, {
    headers: { 'Authorization': authHeader, 'Accept': 'application/json' }
  });
  if (!response.ok) throw new Error(`GET failed: ${response.statusText}`);
  
  const data = await response.json();
  return data.result.map(record => record.sys_id);
}

// 2. Update each record
async function updateRecords(sysIds) {
  for (const sysId of sysIds) {
    const url = `https://${INSTANCE}.service-now.com/api/now/table/${TABLE}/${sysId}`;
    const payload = {
      short_description: 'Updated via External NodeJS API',
      state: '2' // In Progress
    };

    const response = await fetch(url, {
      method: 'PATCH',
      headers: {
        'Authorization': authHeader,
        'Content-Type': 'application/json',
        'Accept': 'application/json'
      },
      body: JSON.stringify(payload)
    });

    if (response.ok) {
      console.log(`Updated ${sysId} successfully.`);
    }
  }
}

// Execute
(async () => {
  try {
    const sysIds = await fetchSysIds();
    console.log(`Found ${sysIds.length} records.`);
    await updateRecords(sysIds);
  } catch (error) {
    console.error('Error:', error.message);
  }
})();

2. External Code Search Integration (sn_codesearch API)​

ServiceNow Studio has a feature called "Code Search" that allows developers to find keywords across Business Rules, Script Includes, Client Scripts, etc. You can hook into this directly via the sn_codesearch REST API.

This is extremely useful for running automated security scans or finding hardcoded sys_ids from your local machine.

APIs Used:

  • GET /api/sn_codesearch/code_search/search?term={keyword}
javascript
import https from 'https';

const keyword = 'TODO'; // Search term
const INSTANCE = 'your_instance';
const USERNAME = 'admin_user';
const PASSWORD = 'admin_password';

function codeSearch(keyword) {
    return new Promise((resolve, reject) => {
        const uri = `https://${INSTANCE}.service-now.com/api/sn_codesearch/code_search/search?term=${keyword}`;

        const options = {
            method: 'GET',
            headers: {
                'Authorization': `Basic ${Buffer.from(`${USERNAME}:${PASSWORD}`).toString('base64')}`,
                'Accept': 'application/json'
            }
        };

        const req = https.request(uri, options, (res) => {
            let data = '';
            res.on('data', (chunk) => { data += chunk; });
            res.on('end', () => {
                try {
                    const json = JSON.parse(data);
                    if (json.result && json.result.length > 0) {
                        // Filter out tables that returned 0 hits
                        json.result = json.result.filter(item => item.hits.length > 0);
                        resolve(json);
                    } else {
                        resolve(null);
                    }
                } catch (e) {
                    reject(e);
                }
            });
        });
        req.on('error', reject);
        req.end();
    });
}

(async () => {
    console.log(`Performing ServiceNow Code Search for Keyword: '${keyword}'...`);
    try {
        const result = await codeSearch(keyword);
        if (result) {
            console.log(`Found results in ${result.result.length} tables.`);
            console.log(JSON.stringify(result, null, 2));
        } else {
            console.log(`No results found.`);
        }
    } catch (error) {
        console.error(`Error: ${error.message}`);
    }
})();
