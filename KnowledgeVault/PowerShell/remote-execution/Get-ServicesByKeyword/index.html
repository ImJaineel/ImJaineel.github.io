Get-ServicesByKeyword

Synopsis​

Retrieves Windows services matching a keyword from multiple remote servers.

Description​

Connects to each server in the provided list via WinRM and queries Win32_Service for services whose Name, DisplayName, or Description matches the given keyword. Outputs a formatted table per server including: Name, Display Name, Logon As, State, and Description. Continues to the next server on failure.

Parameters​

User​

Username for WinRM authentication.

UserPassword​

Plaintext password for authentication (converted to SecureString internally).

Servers​

Array of server hostnames or IPs to connect to.

Keyword​

Keyword string to filter services by. Defaults to "MID".

Examples​

Example 1​

powershell
.\Get-ServicesByKeyword.ps1 -User "domain\admin" -UserPassword "pass" -Servers "srv01","srv02" -Keyword "MID"

Script Code​

Get-ServicesByKeyword.ps1
powershell
<#
.SYNOPSIS
    Retrieves Windows services matching a keyword from multiple remote servers.
.DESCRIPTION
    Connects to each server in the provided list via WinRM and queries Win32_Service
    for services whose Name, DisplayName, or Description matches the given keyword.
    Outputs a formatted table per server including: Name, Display Name, Logon As,
    State, and Description. Continues to the next server on failure.
.PARAMETER User
    Username for WinRM authentication.
.PARAMETER UserPassword
    Plaintext password for authentication (converted to SecureString internally).
.PARAMETER Servers
    Array of server hostnames or IPs to connect to.
.PARAMETER Keyword
    Keyword string to filter services by. Defaults to "MID".
.EXAMPLE
    .\Get-ServicesByKeyword.ps1 -User "domain\admin" -UserPassword "pass" -Servers "srv01","srv02" -Keyword "MID"
.TAGS
    service, wmi, win32_service, keyword, filter, remote, winrm, table, report
.NOTES
    Author: Jaineel Petiwale
#>
param (
    [string]$User,
    [string]$UserPassword,
    [string[]]$Servers,
    [string]$Keyword = "MID"
)

$secureStringPwd = ConvertTo-SecureString $UserPassword -AsPlainText -Force
$credential = New-Object System.Management.Automation.PSCredential -ArgumentList $User, $secureStringPwd

Write-Host "`n# $Keyword Windows Services Search Report — $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss') $((Get-TimeZone).Id)`n---" -ForegroundColor Green

foreach ($server in $Servers) {
    Write-Host "## $server`n" -ForegroundColor Yellow
    try {
        Invoke-Command -ComputerName $server -Credential $credential -Authentication Negotiate -ErrorAction Stop -ScriptBlock {
            $hostname = $env:COMPUTERNAME
            $keyword  = $using:Keyword

            $services = Get-WmiObject Win32_Service | Where-Object {
                $_.Name        -match $keyword -or
                $_.DisplayName -match $keyword -or
                $_.Description -match $keyword
            } | Sort-Object State | ForEach-Object {
                [PSCustomObject]@{
                    Name        = $_.Name
                    DisplayName = $_.DisplayName
                    LogonAs     = $_.StartName
                    State       = $_.State
                    Description = $_.Description
                }
            }

            $maxName        = [Math]::Max(($services.Name        | Measure-Object -Property Length -Maximum).Maximum, 25)
            $maxDisplayName = [Math]::Max(($services.DisplayName | Measure-Object -Property Length -Maximum).Maximum, 30)
            $maxLogonAs     = [Math]::Max(($services.LogonAs     | Measure-Object -Property Length -Maximum).Maximum, 30)
            $maxState       = [Math]::Max(($services.State       | Measure-Object -Property Length -Maximum).Maximum, 10)
            $maxDescription = [Math]::Max(($services.Description | Measure-Object -Property Length -Maximum).Maximum, 40)

            Write-Host "**Hostname**: ``$hostname```n" -ForegroundColor Cyan

            if ($services.Count -eq 0) {
                Write-Host "No '$keyword' services found on $hostname." -ForegroundColor DarkYellow
                return
            }

            Write-Host ("{0,-$maxName} | {1,-$maxDisplayName} | {2,-$maxLogonAs} | {3,-$maxState} | {4,-$maxDescription}" `
                -f "Name", "Display Name", "Logon As", "State", "Description")
            Write-Host ("{0,-$maxName} | {1,-$maxDisplayName} | {2,-$maxLogonAs} | {3,-$maxState} | {4,-$maxDescription}" `
                -f ("-" * $maxName), ("-" * $maxDisplayName), ("-" * $maxLogonAs), ("-" * $maxState), ("-" * $maxDescription))

            foreach ($svc in $services) {
                Write-Host ("{0,-$maxName} | {1,-$maxDisplayName} | {2,-$maxLogonAs} | {3,-$maxState} | {4,-$maxDescription}" `
                    -f $svc.Name, $svc.DisplayName, $svc.LogonAs, $svc.State, $svc.Description)
            }
        }
    } catch {
        Write-Warning "Failed to connect to $server : $_"
    }
    Write-Host "`n---"
}
